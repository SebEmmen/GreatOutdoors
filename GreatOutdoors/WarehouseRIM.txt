DROP TABLE IF EXISTS Inventory_Forecast_Target;
DROP TABLE IF EXISTS Satisfaction;
DROP TABLE IF EXISTS Training;
DROP TABLE IF EXISTS Returned_Item;      -- References Orders
DROP TABLE IF EXISTS Orders;             -- References Product, Sites, Sales, Staff
DROP TABLE IF EXISTS Customer_Contact;   -- References Sites
DROP TABLE IF EXISTS Sales_Demographic;  -- References Customer_Headquarters
DROP TABLE IF EXISTS Staff;              -- References Sales
DROP TABLE IF EXISTS Product;
DROP TABLE IF EXISTS Sites;              -- References Customer
DROP TABLE IF EXISTS Customer;           -- References Customer_Headquarters
DROP TABLE IF EXISTS Sales;
DROP TABLE IF EXISTS Customer_Headquarters;


-- 1. Parent table for the entire hierarchy
CREATE TABLE Customer_Headquarters (
    customer_codemr INT,
    customer_name VARCHAR(100),
    address1 VARCHAR(100),
    address2 VARCHAR(45),
    city VARCHAR(45),
    region VARCHAR(45),
    postal_zone TEXT,
    country_code INT,
    phone TEXT,
    fax TEXT,
    segment_code INT,
    language CHAR(3),
    segment_name VARCHAR(45),
    segment_description VARCHAR(100),
    PRIMARY KEY (customer_codemr)
);

-- 2. Customer table (linked to Headquarters)
CREATE TABLE Customer (
    customer_code INT,
    customer_codemr INT,
    company_name VARCHAR(45),
    customer_type VARCHAR(45),
    PRIMARY KEY (customer_code),
    FOREIGN KEY (customer_codemr) REFERENCES Customer_Headquarters(customer_codemr)
);

-- 3. Store locations for specific Customers
CREATE TABLE Sites (
    site_skey INTEGER PRIMARY KEY AUTOINCREMENT,
    site_code INT,
    customer_code INT,
    street VARCHAR(100),
    addition TEXT,
    city VARCHAR(45),
    state VARCHAR(45),
    postal_zone TEXT,
    country_code INT,
    active_indicator INT,
    country VARCHAR(45),
    flag_image TEXT,
    currency_name VARCHAR(45),
    territory_name VARCHAR(45),
    is_current INT,
    valid_from DATE,
    valid_to DATE,
    FOREIGN KEY (customer_code) REFERENCES Customer(customer_code)
);

-- 4. Contact persons for specific Stores
CREATE TABLE Customer_Contact (
    customer_contact_code INT,
    site_code INT,
    first_name VARCHAR(45),
    last_name VARCHAR(45),
    job_position VARCHAR(100),
    extension TEXT,
    fax TEXT,
    e_mail TEXT,
    gender CHAR(1),
    PRIMARY KEY (customer_contact_code),
    FOREIGN KEY (site_code) REFERENCES Customer_Store(site_skey)
);

-- 5. Demographic data for Headquarters
CREATE TABLE Sales_Demographic (
    demographic_code INT,
    customer_codemr INT,
    sales_percent INT,
    upper_age INT,
    lower_age INT,
    PRIMARY KEY (demographic_code),
    FOREIGN KEY (customer_codemr) REFERENCES Customer_Headquarters(customer_codemr)
);

-- 6. Sales Branch data
CREATE TABLE Sales (
	sale_skey INTEGER PRIMARY KEY AUTOINCREMENT,
	sale_code INT,
	street VARCHAR(100),
	addition TEXT,
	city VARCHAR(45),
	state VARCHAR(45),
	postal_zone TEXT,
	country VARCHAR(45),
	currency_name VARCHAR(45),
	is_current INT,
    	valid_from DATE,
    	valid_to DATE
);

-- 7. Sales Staff
CREATE TABLE Staff (
	staff_skey INTEGER PRIMARY KEY AUTOINCREMENT,
	staff_code INT,
	first_name VARCHAR(45),
	last_name VARCHAR(45),
	position VARCHAR(100),
	work_phone TEXT,
	extension TEXT,
	fax TEXT,
	email TEXT,
	date_hired DATE,
	is_current INT,
    	valid_from DATE,
    	valid_to DATE,
	sales_code INT,
	manager_code INT,
	FOREIGN KEY(sales_code) REFERENCES Sales(sale_code)
);

-- 8. Product
CREATE TABLE Product(
	product_number INT,
	introduction_date DATE,
	product_image TEXT,
	product_name VARCHAR(45),
	description VARCHAR(100),
	product_type VARCHAR(45),
	product_line VARCHAR(45),
	PRIMARY KEY(product_number)

);

-- 9. Orders
CREATE TABLE Orders(
	order_code INT,
	order_number INT,
	product_number INT,
	total_revenue FLOAT,
	total_cost FLOAT,
	gross_profit FLOAT,
	profit_margin FLOAT,
	retailer_name VARCHAR(45),
	site_code INT,
	retailer_contact_code INT,
	staff_code INT,
	sales_code INT,
	order_date DATE,
	order_method VARCHAR(45),
	PRIMARY KEY(order_code),
	FOREIGN KEY(product_number) REFERENCES Product(product_number),
	FOREIGN KEY(site_code) REFERENCES Sites(site_skey),
	FOREIGN KEY(sales_code) REFERENCES Sales(sale_skey),
	FOREIGN KEY(staff_code) REFERENCES Staff(staff_skey)	
);

-- 10. Returned_Item
CREATE TABLE Returned_Item(
	return_code INT,
	return_date DATE,
	order_code INT,
	return_quantity INT,
	return_description VARCHAR(100),
	PRIMARY KEY(return_code),
	FOREIGN KEY(order_code) REFERENCES Orders(order_code)
);

-- 11. Training
CREATE TABLE Training (
    year INT NOT NULL,
    staff_code INT NOT NULL,
    course_description VARCHAR(255),
    
    -- Composite Primary Key to ensure uniqueness across the three identifying columns
    PRIMARY KEY (year, staff_code),
    FOREIGN KEY(staff_code) REFERENCES Staff(staff_skey)
);

-- 12. Satisfaction
CREATE TABLE Satisfaction(
	year INT NOT NULL,
	staff_code INT NOT NULL,
	satisfaction_type_description VARCHAR(100),

	PRIMARY KEY (year, staff_code),
        FOREIGN KEY(staff_code) REFERENCES Staff(staff_skey)
);

CREATE TABLE Inventory_Forecast_Target (
	year INT,
	month INT,
	product_number INT,
	retailer_name VARCHAR(45),
	staff_code INT,
	sales_target INT,
	inventory_count INT,
	expected_volume INT,
	retailer_code INT,
	PRIMARY KEY(year, product_number, month),
	FOREIGN KEY(product_number) REFERENCES Product(product_number),
	FOREIGN KEY(staff_code) REFERENCES Staff(staff_skey),
	FOREIGN KEY(retailer_code) REFERENCES Site(customer_code)	
);
